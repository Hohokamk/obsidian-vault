双指针法
int cmp(const triple &a, const triple &b) {
    if (a.row != b.row) return a.row - b.row;
    return a.col - b.col;
}
Matrix addition_sorted(const Matrix &A, const Matrix &B) {
    Matrix C;
    C.rows = A.rows;
    C.cols = A.cols;
    C.counts = 0;
    int i = 0, j = 0;
    while (i < A.counts && j < B.counts) {
        int c = cmp(A.data[i], B.data[j]);
        if (c < 0) {                    // A 的元素位置更靠前
            C.data[C.counts++] = A.data[i++];
        } else if (c > 0) {             // B 的更靠前
            C.data[C.counts++] = B.data[j++];
        } else {                        // 位置相同 → 相加
            int v = A.data[i].value + B.data[j].value;
            if (v != 0) {               // 结果为 0 则不存
                C.data[C.counts++] = {A.data[i].row, A.data[i].col, v};
            }
            i++; j++;
        }
    }
    // 收尾：把剩余部分直接搬过来
    while (i < A.counts) C.data[C.counts++] = A.data[i++];
    while (j < B.counts) C.data[C.counts++] = B.data[j++];
    return C;
}
