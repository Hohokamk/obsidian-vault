void ControlTask(void* pvParameters)
{
    while (true)
    {
        // rc.ch[0] = -660~660 → 映射到编码器 0~8191
        int32_t target = (rc.ch[0] + 660) * 8191 / 1320;
        can2_motor[0].setangle = target;

        ctrl.chassis.Update();
        rc.Update();
        vTaskDelay(5);
    }
}
