void StartTask 03(void *argument)
{
  /* USER CODE BEGIN StartTask 03 */
  uint 8_t cmd;
  uint 8_t led_index = 0;
  GPIO_TypeDef* led_ports[] = {GPIOC, GPIOC, GPIOC, GPIOC};
  uint 16_t led_pins[] = {GPIO_PIN_ 0, GPIO_PIN_ 1, GPIO_PIN_ 2, GPIO_PIN_ 3};
  
  for(;;)
  {
    // 等待队列消息
    if(osMessageQueueGet(uartQueueHandle, &cmd, NULL, osWaitForever) == osOK)
    {
      if(cmd == 1)  // yes - 启动流水灯
      {
        while(1)
        {
          // 熄灭所有 LED
          for(int i = 0; i < 4; i++)
          {
            HAL_GPIO_WritePin(led_ports[i], led_pins[i], GPIO_PIN_RESET);
          }
          
          // 点亮当前 LED
          HAL_GPIO_WritePin(led_ports[led_index], led_pins[led_index], GPIO_PIN_SET);
          led_index = (led_index + 1) % 4;
          
          // 延时（可被状态机改变）
          osDelay(g_flow_delay);
          
          // 检查是否有新命令（非阻塞）
          if(osMessageQueueGet(uartQueueHandle, &cmd, NULL, 0) == osOK)
          {
            if(cmd == 0)  // no - 停止流水灯
            {
              // 熄灭所有 LED
              for(int i = 0; i < 4; i++)
              {
                HAL_GPIO_WritePin(led_ports[i], led_pins[i], GPIO_PIN_RESET);
              }
              break;  // 跳出流水灯循环
            }
          }
        }
      }
      else  // no - 直接熄灭
      {
        for(int i = 0; i < 4; i++)
        {
          HAL_GPIO_WritePin(led_ports[i], led_pins[i], GPIO_PIN_RESET);
        }
      }
    }
  }
  /* USER CODE END StartTask 03 */
}
