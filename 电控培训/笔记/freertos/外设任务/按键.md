void StartTask05(void *argument)
{
  /* USER CODE BEGIN StartTask05 */
  uint8_t key8_last = 1;  // PC8 上一次状态（1=未按下）
  uint32_t key9_press_time = 0;
  uint8_t key9_pressed = 0;
  
  for(;;)
  {
    // === PC8：状态机切换（短按） ===
    uint8_t key8_now = HAL_GPIO_ReadPin(GPIOC, GPIO_PIN_8);
    
    if(key8_last == 1 && key8_now == 0)  // 下降沿
    {
      osDelay(20);  // 消抖
      if(HAL_GPIO_ReadPin(GPIOC, GPIO_PIN_8) == 0)
      {
        // 切换状态
        g_motor_state = (g_motor_state + 1) % 3;
        
        // 根据状态设置参数
        switch(g_motor_state)
        {
          case 0:  // STOP
            g_flow_delay = 500;
            g_motor_pwm = 0;
            printf("State: STOP\r\n");
            break;
          case 1:  // RUN
            g_flow_delay = 200;
            g_motor_pwm = 500;
            printf("State: RUN\r\n");
            break;
          case 2:  // FAST
            g_flow_delay = 80;
            g_motor_pwm = 999;
            printf("State: FAST\r\n");
            break;
        }
        
        // 更新电机 PWM
        __HAL_TIM_SET_COMPARE(&htim2, TIM_CHANNEL_3, g_motor_pwm);
        
        // 等待释放
        while(HAL_GPIO_ReadPin(GPIOC, GPIO_PIN_8) == 0);
      }
    }
    key8_last = key8_now;
    
    // === PC9：长按 2s 恢复 ===
    uint8_t key9_now = HAL_GPIO_ReadPin(GPIOC, GPIO_PIN_9);
    
    if(key9_now == 0)  // 按下
    {
      if(!key9_pressed)
      {
        key9_pressed = 1;
        key9_press_time = osKernelGetTickCount();
      }
      else
      {
        // 检查是否超过 2s
        if((osKernelGetTickCount() - key9_press_time) >= 2000)
        {
          osThreadResume(Task_E_SuspendHandle);
          printf("Task_E_Suspend resumed!\r\n");
          key9_pressed = 0;
          // 等待释放
          while(HAL_GPIO_ReadPin(GPIOC, GPIO_PIN_9) == 0);
        }
      }
    }
    else
    {
      key9_pressed = 0;
    }
    
    osDelay(10);  // 轮询间隔
  }
  /* USER CODE END StartTask05 */
}
