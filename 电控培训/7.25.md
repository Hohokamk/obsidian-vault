{

/ _前 6 字节全部为 0 时，不进行解码_ /

if ((m_frame[0] | m_frame[1] | m_frame[2] |

m_frame[3] | m_frame[4] | m_frame[5]) == 0)

{

return;

}

```
/* 解码4个11位摇杆通道 */
rc.ch[0] =
    ((m_frame[0] | (m_frame[1] << 8)) & 0x07FF) - 1024;

rc.ch[1] =
    (((m_frame[1] >> 3) | (m_frame[2] << 5)) & 0x07FF) - 1024;

rc.ch[2] =
    (((m_frame[2] >> 6) |
      (m_frame[3] << 2) |
      (m_frame[4] << 10)) & 0x07FF) - 1024;

rc.ch[3] =
    (((m_frame[4] >> 1) |
      (m_frame[5] << 7)) & 0x07FF) - 1024;

/* 摇杆中值死区 */
if (rc.ch[0] >= -8 && rc.ch[0] <= 8)
    rc.ch[0] = 0;

if (rc.ch[1] >= -8 && rc.ch[1] <= 8)
    rc.ch[1] = 0;

if (rc.ch[2] >= -8 && rc.ch[2] <= 8)
    rc.ch[2] = 0;

if (rc.ch[3] >= -8 && rc.ch[3] <= 8)
    rc.ch[3] = 0;

/* 解码两个三挡拨杆 */
rc.s[0] = (m_frame[5] >> 6) & 0x03;
rc.s[1] = (m_frame[5] >> 4) & 0x03;
```
