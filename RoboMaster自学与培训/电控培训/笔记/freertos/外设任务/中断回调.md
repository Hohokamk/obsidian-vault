void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart)
{
  if(huart->Instance == USART 1)
  {
    // 收到一个字节
    if(uart_rx_index < sizeof(uart_rx_buffer) - 1)
    {
      uart_rx_buffer[uart_rx_index++] = uart_rx_byte;
    }
    
    // 检测到回车或换行
    if(uart_rx_byte == '\r' || uart_rx_byte == '\n')
    {
      uart_rx_buffer[uart_rx_index - 1] = '\0';  // 去掉回车符
      uart_rx_index = 0;
      
      // 判断命令
      uint8_t cmd = 0;  // 0=no, 1=yes
      if(strcmp(uart_rx_buffer, "yes") == 0)
      {
        cmd = 1;
        printf("Received: yes\r\n");
      }
      else if(strcmp(uart_rx_buffer, "no") == 0)
      {
        cmd = 0;
        printf("Received: no\r\n");
      }
      
      // 发送到队列
      osMessageQueuePut(uartQueueHandle, &cmd, 0, 0);
    }
    
    // 重新启动中断接收
    HAL_UART_Receive_IT(&huart1, &uart_rx_byte, 1);
  }
}
