## 完整代码结构拆解

### 1. 初始化部分 —— "起床仪式"

```c
// main.c 开头，所有东西启动前必须执行

void System_Init(void) {
    // 1. 系统时钟配置（STM32 的心脏起搏器）
    // 比喻：就像你起床后先让心脏正常跳动
    SystemClock_Config();  // 把 8MHz 晶振倍频到 72MHz
    
    // 2. GPIO 初始化（手脚的神经连接）
    // 比喻：让大脑能控制手脚
    GPIO_Init();  // 配置哪些引脚是输出（控制电机方向）、哪些是输入（读传感器）
    
    // 3. 定时器初始化（心跳节拍）
    // 比喻：设定一个闹钟，每 10ms 响一次，提醒你该做一件事了
    TIM2_Init();  // 编码器输入（读轮子转了多少）
    TIM3_Init();  // 10ms 中断（控制周期）
    TIM4_Init();  // PWM 输出（控制电机转速）
    
    // 4. 通信初始化（和外界说话的能力）
    // 比喻：戴上眼镜（OLED）、连上耳机（串口）
    I2C_Init();   // 接 MPU6050
    OLED_Init();  // 显示屏
    UART_Init();  // 串口调试
    
    // 5. 外设初始化
    MPU6050_Init();  // 陀螺仪"归零"
    Encoder_Init();  // 编码器计数清零
    Motor_Stop();    // 电机先停下，别乱动
}
```

**形象理解**：就像你早上起床，先睁眼（GPIO）、让心脏正常跳（时钟）、设定每 10 分钟看一次时间的节奏（定时器）、戴上眼镜（OLED）、活动一下手脚确认能动（电机测试）。

### 2. 传感器读取 —— "眼睛和皮肤"

```c
// 编码器读取：数轮子转了多少圈
// 比喻：就像你走路时心里默数"一步、两步、三步..."
int16_t Read_Encoder_Left(void) {
    int16_t count = TIM2->CNT;  // 读取定时器计数器的值
    TIM2->CNT = 0;              // 清零，下次重新数
    return count;               // 返回这 10ms 内走了多少步
}

// MPU6050 读取：感知"我现在面朝哪个方向"
// 比喻：就像你闭着眼睛也能感觉到自己在转圈
float Read_MPU_Yaw(void) {
    MPU6050_Read_Data();           // 读原始数据
    yaw += gyro_z * dt;            // 角速度 × 时间 = 转过的角度
    // 或者用 DMP 直接输出姿态四元数
    return yaw;
}

// 循迹传感器读取：看黑线在哪里
// 比喻：就像你低头看脚下的路，看自己在路的左边还是右边
int8_t Read_Trace_Sensors(void) {
    // 5 路传感器，0=白，1=黑
    uint8_t s0 = HAL_GPIO_ReadPin(SENSOR0_GPIO, SENSOR0_PIN);
    uint8_t s1 = HAL_GPIO_ReadPin(SENSOR1_GPIO, SENSOR1_PIN);
    uint8_t s2 = HAL_GPIO_ReadPin(SENSOR2_GPIO, SENSOR2_PIN);
    uint8_t s3 = HAL_GPIO_ReadPin(SENSOR3_GPIO, SENSOR3_PIN);
    uint8_t s4 = HAL_GPIO_ReadPin(SENSOR4_GPIO, SENSOR4_PIN);
    
    // 计算偏差：黑线在哪？中间是 0，偏左是负，偏右是正
    if(s2) return 0;       // 中间在黑线上 → 正好
    if(s1) return -1;      // 偏左一点
    if(s3) return +1;      // 偏右一点
    if(s0) return -2;      // 偏左很多
    if(s4) return +2;      // 偏右很多
    return 0;              // 全白 → 丢线，保持上次状态
}
```

**形象理解**：
- 编码器 = 你走路默数步数
- MPU 6050 = 你内耳的前庭系统，感知转向
- 循迹传感器 = 你低头看脚下的跑道线

### 3. PID 控制算法 —— "大脑思考"

```c
// PID 结构体：记录一个 PID 控制器的"记忆"
typedef struct {
    float Kp, Ki, Kd;      // 三个参数
    float error;            // 现在差多少
    float last_error;       // 上次差多少
    float integral;         // 累积误差（记忆）
    float output;           // 最终输出
} PID_TypeDef;

// PID 计算过程
// 比喻：你想把水杯倒到 50% 满
// P：现在只有 30%，差 20% → 多倒点
// I：倒了很久还是 45% → 慢慢再加一点
// D：水位上升很快，快 50% 了 → 提前收手
float PID_Calculate(PID_TypeDef *pid, float target, float actual) {
    // 1. 算误差：我想怎样 vs 实际怎样
    pid->error = target - actual;  // 差 20% → error = 20
    
    // 2. 积分：把过去的误差累积起来
    pid->integral += pid->error;    // 如果一直差 5%，积分会越来越大
    
    // 3. 微分：看误差变化的趋势
    float derivative = pid->error - pid->last_error;  // 误差在变大还是变小？
    
    // 4. 三者加权求和
    pid->output = pid->Kp * pid->error           // 比例：现在差多少
                + pid->Ki * pid->integral         // 积分：过去累积差多少
                + pid->Kd * derivative;           // 微分：变化趋势
    
    // 5. 限制输出（不能倒超过 100% 的水）
    if(pid->output > 100) pid->output = 100;
    if(pid->output < -100) pid->output = -100;
    
    // 6. 记录这次误差，下次用
    pid->last_error = pid->error;
    
    return pid->output;
}
```

**形象理解**：
- **Kp（比例）** = 急性子：差得多就猛干，简单粗暴
- **Ki（积分）** = 慢性子：发现总是差一点，慢慢补上
- **Kd（微分）** = 预见性：看到快达标了，提前刹车

**三个 PID 环在小车里**：
```
┌─────────────────────────────────────────┐
│  速度环 PID：目标速度 vs 实际速度          │
│  → 输出：PWM 占空比（给多少"油门"）       │
├─────────────────────────────────────────┤
│  转向环 PID：目标角度 vs 实际角度          │
│  → 输出：左右轮速度差（转多快）            │
├─────────────────────────────────────────┤
│  循迹环 PID：目标位置（黑线中间）vs 实际位置 │
│  → 输出：转向修正量（往哪边拐）            │
└─────────────────────────────────────────┘
```

### 4. 电机驱动 —— "手脚肌肉"

```c
// 控制单个电机
// AIN1=1, AIN2=0 → 正转
// AIN1=0, AIN2=1 → 反转
// PWMA = PWM 占空比（0~100）
void Motor_Left_Set(int16_t speed) {
    if(speed > 0) {  // 正转
        HAL_GPIO_WritePin(AIN1_GPIO, AIN1_PIN, 1);
        HAL_GPIO_WritePin(AIN2_GPIO, AIN2_PIN, 0);
        __HAL_TIM_SET_COMPARE(&htim4, TIM_CHANNEL_1, speed);  // PWM 输出
    } else if(speed < 0) {  // 反转
        HAL_GPIO_WritePin(AIN1_GPIO, AIN1_PIN, 0);
        HAL_GPIO_WritePin(AIN2_GPIO, AIN2_PIN, 1);
        __HAL_TIM_SET_COMPARE(&htim4, TIM_CHANNEL_1, -speed);
    } else {  // 停止
        HAL_GPIO_WritePin(AIN1_GPIO, AIN1_PIN, 0);
        HAL_GPIO_WritePin(AIN2_GPIO, AIN2_PIN, 0);
        __HAL_TIM_SET_COMPARE(&htim4, TIM_CHANNEL_1, 0);
    }
}

// 同时控制两个电机（带差速）
// base：基础速度，turn：转向修正（正=右转，负=左转）
void Motor_Dual_Set(int16_t base, int16_t turn) {
    int16_t left = base - turn;   // 要右转 → 左轮快，右轮慢
    int16_t right = base + turn;  // 要左转 → 右轮快，左轮慢
    
    Motor_Left_Set(left);
    Motor_Right_Set(right);
}
```

**形象理解**：
- TB 6612 = 你的脊髓，接收大脑指令，控制肌肉收缩
- PWM = 肌肉收缩的力度（轻轻推 vs 使劲推）
- 方向引脚 = 肌肉收缩的方向（屈 vs 伸）

### 5. 状态机 —— "任务规划大脑"

```c
// 定义所有状态
typedef enum {
    STATE_IDLE,         // 待机
    STATE_GO_STRAIGHT,  // 直走
    STATE_TURN_LEFT,    // 左转弧
    STATE_TURN_RIGHT,   // 右转弧
    STATE_STOP,         // 停车
    STATE_BEEP          // 声光提示
} CarState;

CarState current_state = STATE_IDLE;

// 状态机主循环
void State_Machine_Update(void) {
    switch(current_state) {
        
        case STATE_IDLE:
            Motor_Dual_Set(0, 0);  // 不动
            if(Start_Button_Pressed()) {
                current_state = STATE_GO_STRAIGHT;
                target_distance = 1000;  // 目标：走 1000 个脉冲
                encoder_start = Get_Total_Encoder();  // 记录起点
            }
            break;
        
        case STATE_GO_STRAIGHT:
            // 速度环：保持 100 的速度
            // 循迹环：保持在线中间
            speed_out = PID_Calculate(&pid_speed, 100, actual_speed);
            trace_out = PID_Calculate(&pid_trace, 0, trace_error);
            Motor_Dual_Set(speed_out, trace_out);
            
            // 检查：走到目标距离了吗？
            if(Get_Total_Encoder() - encoder_start >= target_distance) {
                Beep();  // 声光提示
                current_state = STATE_TURN_RIGHT;
                target_angle = Get_Yaw() + 180;  // 目标：转 180 度
            }
            break;
        
        case STATE_TURN_RIGHT:
            // 差速转弯：左轮快，右轮慢
            Motor_Dual_Set(80, 40);  // base=80, turn=40 → 右轮=120, 左轮=40
            
            // 检查：转到目标角度了吗？
            if(Get_Yaw() >= target_angle) {
                Beep();
                current_state = STATE_GO_STRAIGHT;
                // ... 更新下一个目标
            }
            break;
        
        case STATE_STOP:
            Motor_Dual_Set(0, 0);
            Beep_Long();  // 长响，表示结束
            break;
    }
}
```

**形象理解**：
- 状态机 = 你早上规划"先刷牙→再吃饭→再出门"
- 每个状态 = 一个具体动作
- 切换条件 = "刷完了吗？""吃饱了吗？"
- 声光提示 = 做完一件事的"打卡"

### 6. 中断/定时器 —— "心跳节拍"

```c
// 定时器中断服务函数，每 10ms 执行一次
// 比喻：就像心跳，每跳一下，身体做一次循环
void TIM3_IRQHandler(void) {
    if(TIM3->SR & TIM_SR_UIF) {  // 更新中断标志
        TIM3->SR &= ~TIM_SR_UIF;  // 清除标志
        
        // 1. 读取所有传感器（看一眼路）
        actual_speed_left = Read_Encoder_Left();
        actual_speed_right = Read_Encoder_Right();
        actual_yaw = Read_MPU_Yaw();
        trace_error = Read_Trace_Sensors();
        
        // 2. 执行 PID 计算（大脑思考）
        // 3. 更新状态机（决定下一步）
        State_Machine_Update();
        
        // 4. 输出到电机（动手脚）
        // （在状态机里已经做了）
        
        // 5. 更新显示（告诉别人我在干嘛）
        OLED_Update();
    }
}
```

**形象理解**：
- 10 ms = 100 Hz = 每秒心跳 100 次
- 每次心跳 = 看路 → 思考 → 动手 → 显示
- 为什么用中断？因为主循环 `while(1)` 可能在做其他事（比如显示），中断保证"看路-思考-动手"的节奏绝对稳定，不被打断

### 7. 主循环 —— "后台杂事"

```c
int main(void) {
    System_Init();  // 起床仪式
    
    while(1) {
        // 中断在后台每 10ms 做控制
        // 主循环做不那么急的事：
        
        // 1. 显示信息（刷新屏幕）
        OLED_ShowString(0, 0, "Speed L:");
        OLED_ShowNumber(50, 0, actual_speed_left);
        OLED_ShowString(0, 2, "Yaw:");
        OLED_ShowFloat(50, 2, actual_yaw);
        OLED_ShowString(0, 4, "State:");
        OLED_ShowString(50, 4, State_Name[current_state]);
        
        // 2. 串口打印调试信息
        printf("State:%d Speed:%d Yaw:%.2f\n", 
               current_state, actual_speed_left, actual_yaw);
        
        // 3. 按键扫描（检测是否按了启动键）
        Key_Scan();
        
        HAL_Delay(50);  // 主循环 50ms 刷新一次，不抢中断的资源
    }
}
```

**形象理解**：
- 中断 = 你的心跳和呼吸（自动的，不能停）
- 主循环 = 你闲下来刷手机、和别人聊天（不重要，可以慢一点）
